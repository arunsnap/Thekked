// Configure local PostgreSQL connection block
datasource db {
  provider = "postgresql"
  url      = env("DATABASE_URL")
}

generator client {
  provider = "prisma-client-js"
}

model User {
  id            String    @id @default(uuid())
  email         String    @unique
  passwordHash  String
  name          String?
  role          Role      @default(CUSTOMER)
  createdAt     DateTime  @default(now())
  updatedAt     DateTime  @updatedAt
  orders        Order[]
  wishlist      Product[] @relation("UserWishlist")
}

model Product {
  id                String       @id @default(uuid())
  title             String
  brand             String
  price             Decimal      @db.Decimal(10, 2)
  retailPrice       Decimal?     @db.Decimal(10, 2)
  description       String       @db.Text
  era               String       // e.g., "80s", "90s", "Y2K"
  conditionGrade    Condition    @default(EXCELLENT)
  flawDetails       String?      @db.Text
  measurements      Json         // format: { chestInches: 21, lengthInches: 28, sleeveInches: 33 }
  fabricComposition String?
  imageUrl          String
  isSold            Boolean      @default(false)
  createdAt         DateTime     @default(now())
  updatedAt         DateTime     @updatedAt
  orderId           String?
  order             Order?       @relation(fields: [orderId], references: [id])
  wishlistedBy      User[]       @relation("UserWishlist")
}

model Order {
  id                String       @id @default(uuid())
  orderNumber       String       @unique @default(cuid())
  userId            String
  user              User         @relation(fields: [userId], references: [id])
  products          Product[]    // Dynamic association allows multi-item carts, but locks products
  totalPrice        Decimal      @db.Decimal(10, 2)
  stripePaymentId   String       @unique
  shippingAddress   Json
  fulfillmentStatus Status       @default(PENDING)
  createdAt         DateTime     @default(now())
}

enum Role {
  CUSTOMER
  ADMIN
  CURATOR
}

enum Condition {
  MINT
  EXCELLENT
  GOOD
}

enum Status {
  PENDING
  DISPATCHED
  FULFILLED
  REFUNDED
}
